def write_navx(zip_bytes, out_path):
    header = (b"NAVX" + struct.pack("<I", 40) + struct.pack("<I", 2)
              + uuid.UUID(APP_ID).bytes_le + struct.pack("<Q", len(zip_bytes)) + b"NAVX")
    with open(out_path, "wb") as f:
        f.write(header); f.write(zip_bytes)
