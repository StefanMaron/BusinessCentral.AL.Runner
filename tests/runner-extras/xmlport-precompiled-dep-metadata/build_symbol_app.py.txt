import zipfile, io, struct, uuid, sys, json

APP_ID = "d5e6f7a8-9b0c-4d1e-8f2a-3b4c5d6e7f92"
NAME, PUBLISHER, VERSION = "XPD Precompiled XmlPort Dep", "AL Runner Fixtures", "1.0.0.0"

navx_manifest = f'''<?xml version="1.0" encoding="utf-8"?>
<Package xmlns="http://schemas.microsoft.com/navx/2015/manifest">
  <App Id="{APP_ID}" Name="{NAME}" Publisher="{PUBLISHER}" Version="{VERSION}"
       Target="Cloud" ShowMyCode="true" PropagateDependencies="false" />
  <IdRanges><IdRange MinObjectId="61600" MaxObjectId="61699" /></IdRanges>
  <Dependencies />
</Package>
'''

# NO Platform/Runtime attributes and no top-level RuntimeVersion -- #2124: a real value here
# makes BC 27.0's compiler refuse EVERY OTHER dependency .app in the same combined
# tests/runner-extras run with AL1153.
#
# The XmlPorts entry carries Id + Name + Properties and NO node schema, because that is what
# Microsoft's own Base Application SymbolReference.json carries. Verified on BC
# 28.1.49838.53910: its top-level "XmlPorts" is [] and all 40 of its xmlports sit nested under
# "Namespaces", each with exactly the keys
# ['Variables', 'Methods', 'ReferenceSourceFileName', 'Properties', 'Id', 'Name'].
# Both placements are read by the runner's symbol walk; top level is used here because this
# fixture declares no namespace.
symbol_reference = {
    "Tables": [
        {
            "Id": 61600, "Name": "XPDDep Header",
            "Fields": [
                {"Id": 1, "Name": "No.", "TypeDefinition": {"Name": "Code[20]"}, "Properties": []},
                {"Id": 2, "Name": "Description", "TypeDefinition": {"Name": "Text[100]"}, "Properties": []},
            ],
            "Keys": [{"Name": "PK", "FieldNames": ["No."], "Properties": [{"Name": "Clustered", "Value": "1"}]}],
            "ReferenceSourceFileName": "XpdHeader.Table.al", "Properties": [],
        },
    ],
    "XmlPorts": [
        {
            "Id": 61602, "Name": "XPDDep Export Port",
            "Properties": [{"Name": "Direction", "Value": "Export"},
                           {"Name": "Format", "Value": "Xml"}],
            "ReferenceSourceFileName": "XpdExport.XmlPort.al",
            "Variables": [], "Methods": [],
        },
    ],
    "AppId": APP_ID, "Name": NAME, "Publisher": PUBLISHER, "Version": VERSION,
}

def write_navx(zip_bytes, out_path):
    header = (b"NAVX" + struct.pack("<I", 40) + struct.pack("<I", 2)
              + uuid.UUID(APP_ID).bytes_le + struct.pack("<Q", len(zip_bytes)) + b"NAVX")
    with open(out_path, "wb") as f:
        f.write(header); f.write(zip_bytes)

buf = io.BytesIO()
with zipfile.ZipFile(buf, "w", zipfile.ZIP_DEFLATED) as zf:
    zf.writestr("NavxManifest.xml", navx_manifest)
    zf.writestr("SymbolReference.json", json.dumps(symbol_reference))
    zf.writestr("[Content_Types].xml",
        '<?xml version="1.0" encoding="utf-8"?>'
        '<Types xmlns="http://schemas.openxmlformats.org/package/2006/content-types">'
        '<Default Extension="xml" ContentType="" /><Default Extension="json" ContentType="" />'
        '<Default Extension="al" ContentType="" /><Default Extension="png" ContentType="" />'
        '</Types>')
write_navx(buf.getvalue(), sys.argv[1])
print("wrote", sys.argv[1])
